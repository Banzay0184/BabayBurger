## 🔧 Интеграция звуков с PWA

### Service Worker интеграция
```javascript
// В sw.js
self.addEventListener('message', (event) => {
  if (event.data && event.data.type === 'SOUND_NOTIFICATION') {
    // Делегируем воспроизведение звука основному потоку
    event.waitUntil(
      self.clients.matchAll().then(clients => {
        clients.forEach(client => {
          client.postMessage({
            type: 'PLAY_SOUND',
            soundType: event.data.soundType
          });
        });
      })
    );
  }
});
```

### Основной поток обработка
```javascript
// В operator.html
navigator.serviceWorker.addEventListener('message', (event) => {
  if (event.data.type === 'PLAY_SOUND') {
    // Делегируем воспроизведение звука основному потоку
    window.dispatchEvent(new CustomEvent('playSoundFromSW', {
      detail: { soundType: event.data.soundType }
    }));
  }
});
```
